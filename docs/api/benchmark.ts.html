benchmark.ts

Benchmarking library.

@example

import {Benchmark} from '@fuzdev/fuz_util/benchmark.js'; const bench = new Benchmark({ duration_ms: 5000, warmup_iterations: 5, }); bench .add('slugify', () => slugify(title)) .add('slugify_slower', () => slugify_slower(title)); const results = await bench.run(); console.log(bench.table());
